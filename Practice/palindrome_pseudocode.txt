START

RECEIVE WORD/PHRASE

CONVERT WORD/PHRASE TO LOWERCASE

CREATE AN EMPTY STRING CALLED CLEANED_STRING

FOR EACH CHARACTER IN WORD/PHRASE
    IF CHARACTER IS ALPHANUMERIC
        APPEND CHARACTER TO CLEANED_STRING
    END IF
END FOR

CREATE AN EMPTY STRING CALLED REVERSED_STRING

FOR EACH CHARACTER IN CLEANED_STRING FROM LAST TO FIRST
    APPEND CHARACTER TO REVERSED_STRING
END FOR
s
IF CLEANED_STRING = REVERSED_STRING
    OUTPUT "PALINDROME"
ELSE
    OUTPUT "NOT A PALINDROME"
END IF

END
